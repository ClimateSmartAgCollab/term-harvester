python3 ../../term_harvester.py -a https://raw.githubusercontent.com/FoodOntology/foodon/master/foodon.owl

# After the above -a commands, edit config.yaml to add minus/include entries under the
# foodon source, then re-run -c foodon.  Choose one of the two approaches below.
#
# OWL minus.concepts matches rdfs:label (English) or OWL local name (underscore form,
# e.g. BFO_0000020).  minus excludes a class and its entire subtree; include restores a
# class and its subtree even when an ancestor was excluded.  Blocking a parent only
# prevents its subtree from being visited via that specific parent path — multi-parent
# classes can still enter through any unblocked parent path that reaches them first, so
# they must be listed explicitly in minus even if an ancestor is already excluded.


# --- Option A: Exclude approach ---
# Exclude the non-food branches shown in blue in foodon_classes.jpg.
# Simpler, but does not block additional top-level imports that may enter foodon
# from imported ontologies (e.g. BFO_0000001 entity, BFO_0000017 realizable entity,
# time:TemporalEntity, NCIT_C94729).

sources:
  foodon:
    minus:
      concepts:
        - anatomical entity          # UBERON:0001062 (also causes loss of seed & plant ovule via axioms)
        - plant ovule                # PO:0020003  (multi-parent: also reachable via material entity)
        - seed                       # PO:0009010  (multi-parent: also reachable via material entity)
        - anatomical structure       # UBERON:0000061  (multi-parent: also reachable via material entity)
        - UBERON_0006314             # bodily fluid or substance (subclasses covered by animal substance)
        - UBERON_0000456             # secretion of exocrine gland (subclasses covered by animal substance)
        - BFO_0000020                # characteristic (BFO)
        - COB_0000502                # characteristic (COB) — unnecessary duplication of BFO term
        - information content entity # IAO:0000030
        - plant anatomical entity    # PO:0025131 (covers plant structure)
        - TemporalEntity             # time:TemporalEntity — W3C OWL-Time import
        - agency food product type   # FOODON:03400361 (coverage under animal/plant food product)
        - organism piece             # FOODON:00004328 (subclasses already under animal/plant material)

# Re-process foodon with Option A filters applied
python3 ../../term_harvester.py -c foodon


# --- Option B: Include/whitelist approach (alternate) ---
# Specify only what to keep; everything else is excluded automatically.
# More future-proof: any new top-level ontology imports are blocked by default
# without needing explicit minus entries for each one.
#
# When include.concepts is present, the traversal starts all top-level classes
# as excluded and restores only the listed subtrees.  minus.concepts then
# fine-tunes by blocking specific nodes within those restored subtrees.

sources:
  foodon:
    # Order of include and minus does not matter — both are resolved before traversal.
    include:
      concepts:
        - material entity            # BFO:0000040 — only this subtree is kept
    minus:
      concepts:
        # Items within material entity that should be excluded
        - agency food product type   # FOODON:03400361 (coverage under animal/plant food product)
        - organism piece             # FOODON:00004328 (subclasses already under animal/plant material)
        - anatomical structure       # UBERON:0000061  (multi-parent via material anatomical entity)
        - plant ovule                # PO:0020003  (multi-parent via material anatomical entity)
        - seed                       # PO:0009010  (multi-parent via material anatomical entity)
        - UBERON_0006314             # bodily fluid or substance (subclasses covered by animal substance)
        - UBERON_0000456             # secretion of exocrine gland (subclasses covered by animal substance)

# Re-process foodon with Option B filters applied
python3 ../../term_harvester.py -c foodon



# Build schema.yaml from all sources
python3 ../../term_harvester.py -b
